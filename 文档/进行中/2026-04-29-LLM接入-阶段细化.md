### 0.1 `ai-provider.ts` 新增导出

```typescript
// 新增：通用 LLM 调用（从 consult.ts 提取，去掉 consult 专用逻辑）
export async function chatCompletion(
  config: ResolvedAiProviderConfig,
  messages: Array<{ role: string; content: string }>,
  options?: { temperature?: number; maxTokens?: number }
): Promise<unknown> {
  switch (config.transport) {
    case 'openai-compatible':
      // fetch POST → { model, messages, temperature, max_tokens }
    case 'anthropic':
      // fetch POST → { model, system, messages, max_tokens }
      // 处理 system message 分离
    case 'gemini':
      // fetch POST → { contents, generationConfig }
    case 'qwen':
      // fetch POST → { model, input: { messages }, parameters }
    case 'local':
      // fetch POST → { model, messages, stream: false }
  }
}

// 新增：统一文本提取
export function extractText(
  config: ResolvedAiProviderConfig,
  payload: unknown
): string {
  switch (config.transport) {
    case 'openai-compatible': return payload.choices?.[0]?.message?.content || ''
    case 'anthropic': return payload.content?.[0]?.text || ''
    case 'gemini': return payload.candidates?.[0]?.content?.parts?.[0]?.text || ''
    case 'qwen': return payload.output?.text || payload.output?.choices?.[0]?.message?.content || ''
    case 'local': return payload.message?.content || payload.response || ''
  }
}
```

### 1.3 Prompt 模板（系统消息 + 用户消息分离）

```typescript
function buildClassifyPrompt(params: {
  input: string
  interests?: string[]
  sourceContext?: string | null
}): Array<{ role: string; content: string }> {
  const systemPrompt = [
    '你是AI简报助手的意图识别模块。分析用户输入，返回严格的JSON（不要加markdown代码块）。',
    '',
    '可用意图类型及实体格式：',
    '- create_todo: {"content":"待办简短描述","deadline":"明天/下周/待定"}',
    '- record_thought: {"content":"想法简短摘要"}',
    '- fragmented_thought: {"content":"碎片内容","tags":["标签"]}',
    '- add_interest: {"interests":["领域名称"]}',
    '- remove_interest: {"interests":["领域名称"]}',
    '- set_push_time: {"time":"HH:MM"}',
    '- query_stats: {"period":"week|month|recent"}',
    '- chat_only: {}',
    '- multi: {"intents":[{"type":"意图","entities":{}}]}',
    '',
    '回复建议：1-2句自然口语化中文确认/追问。置信度低时可追问澄清。',
    '当intent=chat_only时仍给出自然回应（如"好的，随时找我""嗯嗯，我在听"），不要留空。',
    '建议操作：1-2个上下文相关的后续建议，格式：[{"label":"查看本周AI简报","action":"查看本周AI简报","targetIntent":"query_stats"}]。',
  ].join('\n')

  const userParts: string[] = []
  if (params.interests?.length) {
    userParts.push(`用户当前已关注：${params.interests.join('、')}`)
  }
  if (params.sourceContext) {
    userParts.push(`来源上下文：${params.sourceContext}`)
  }
  userParts.push(`用户输入：${params.input}`)

  return [
    { role: 'system', content: systemPrompt },
    { role: 'user', content: userParts.join('\n') },
  ]
}
```

### 1.4 主入口

```typescript
export async function classifyIntent(params: {
  input: string
  ruleResult: ParsedIntentResult          // 规则引擎的初步结果（用于日志对比）
  config: ResolvedAiProviderConfig
  context?: {
    interests?: string[]
    sourceContext?: string | null
  }
}): Promise<ChatClassifyResult | null> {
  try {
    const messages = buildClassifyPrompt({
      input: params.input,
      interests: params.context?.interests,
      sourceContext: params.context?.sourceContext,
    })

    const controller = new AbortController()
    const timeout = setTimeout(() => controller.abort(), 3000)

    const payload = await chatCompletion(params.config, messages, {
      temperature: 0.3,  // 分类任务低温，兼顾 replyHint 中文自然度
      maxTokens: 300,
    })

    clearTimeout(timeout)

    const text = extractText(params.config, payload)
    if (!text) return null

    return parseClassifyJSON(text)
  } catch (error) {
    console.warn('LLM classify failed, falling back to rule engine:', error)
    return null  // 返回 null → 调用方降级规则结果
  }
}
```

### 1.5 JSON 解析 + 校验

```typescript
function parseClassifyJSON(text: string): ChatClassifyResult | null {
  // 1. 尝试 ```json ... ``` 代码块提取
  // 2. 尝试直接 JSON.parse（先找第一个 { 到最后一个 }）
  // 3. 失败 → return null
  // 4. 校验 intent 在已知列表中，不在 → 修正为 "chat_only"
  // 5. confidence clamp [0, 1]
  // 6. candidateIntents 去重，不含主 intent 时自动追加
  // 7. replyHint 为空时生成默认值
}
```

### 1.8 验证标准

| 输入 | 规则 matchedBy | LLM 是否触发 | LLM 预期 |
|------|---------------|-------------|----------|
| "帮我关注AI和远程工作" | pattern / 0.85 | 否（pattern + ≥0.88） | — |
| "取消关注" | pattern / 0.6 | **是**（pattern + <0.88，无匹配兴趣词） | remove_interest 追问或修正 |
| "明天下午3点提醒我交周报" | keyword / 0.85 | **是**（keyword < 0.88 阈值） | create_todo + {deadline:"明天"} |
| "最近AI方向有什么好的远程机会" | fuzzy / 0.65 | 是 | query_stats 或 multi |
| "这个方向最近动静大吗" | fallback / 0.5 | 是 | query_stats |
| "你好" | fallback / 0.5 | 是 | chat_only（非 greeting，edge-worker 无 greeting 意图） |
| LLM API 500 | — | 是（触发但失败） | 降级规则结果 |
| 用户未配 API Key | — | 否（userProviderConfig=null） | 全走规则 |

## 阶段二：LLM 自然语言回复

**目标**：用 LLM 生成回复文本，替代 `buildIntentAnalysisText()` 和 `buildAssistantReplyText()` 的硬编码模板。

### 2.1 范围与策略

阶段一已通过 `replyHint` 替代 `intent_analysis` 中的确认文本。阶段二扩展为：执行完成后用 LLM 生成 `execution_result` 的回复。

**但阶段二中 reply 是可选的**——分类是核心价值，回复是锦上添花。阶段一完成后即可上线，阶段二在阶段一稳定后再加。

### 2.2 新增函数（`llm-classify.ts`）

```typescript
export async function generateReply(params: {
  actionType: string
  success: boolean
  resultSummary?: string
  userInput: string
  config: ResolvedAiProviderConfig
}): Promise<string | null> {
  const messages = [
    {
      role: 'system' as const,
      content: [
        '你是AI简报助手，风格简洁友善。',
        '根据执行结果生成1句自然的口语化中文回复。',
        '要求：不超过35字，像朋友聊天。失败时表达歉意并建议重试。',
      ].join(' '),
    },
    {
      role: 'user' as const,
      content: [
        `用户说："${params.userInput}"`,
        `执行动作：${params.actionType}`,
        `结果：${params.success ? '成功' : '失败'}`,
        params.resultSummary ? `详情：${params.resultSummary}` : '',
        '请回复：',
      ].filter(Boolean).join('\n'),
    },
  ]

  try {
    const payload = await chatCompletion(params.config, messages, {
      temperature: 0.3,
      maxTokens: 80,
    })
    const text = extractText(params.config, payload)
    return text?.trim() || null
  } catch {
    return null  // 降级模板文本
  }
}
```

### 3.2 `getRecentChatMessages()` — `store.ts` 新增

```typescript
export async function getRecentChatMessages(
  db: D1Database,
  sessionId: number,
  limit: number
): Promise<Array<{ role: string; content: string; intentType: string | null }>> {
  const rows = await queryAll<{ role: string; content: string; intent_type: string | null }>(
    db,
    `SELECT role, content, intent_type FROM chat_messages
     WHERE session_id = ? AND role IN ('user', 'assistant')
     ORDER BY created_at DESC, id DESC LIMIT ?`,
    [sessionId, limit]
  )
  return rows.reverse()  // 按时间正序
}
```

### 4.2 调用日志（可选 D1 表）

```sql
CREATE TABLE IF NOT EXISTS llm_call_logs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  call_type TEXT NOT NULL,       -- 'classify' | 'reply'
  provider TEXT NOT NULL,
  model TEXT NOT NULL,
  duration_ms INTEGER,
  success INTEGER NOT NULL,
  error_message TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now'))

```

### 4.3 监控指标

| 指标 | 计算方式 | 用途 |
|------|----------|------|
| LLM 分类占比 | `llmClassified / totalMessages` | 衡量规则引擎的有效覆盖率 |
| LLM 可用率 | `successfulCalls / attemptedCalls` | Provider 稳定性监控 |
| 分类准确率 | `reclassifyCount / llmClassifiedCount` | 用户纠偏越少 = LLM 越准 |
| 平均延迟 | AVG(duration_ms) per provider | 选择更快的 provider |
